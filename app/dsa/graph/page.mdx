# 15. Detect Cycle (Undirected)

```js
function dfs(node, parent) {

    visited.add(node);

    for (const nei of graph.get(node)) {

        if (!visited.has(nei)) {

            if (dfs(nei, node))
                return true;

        }

        else if (nei !== parent)

            return true;

    }

    return false;

}
```

# 16. Detect Cycle (Directed)

```js
const visited = new Set();
const path = new Set();
```

```js
function dfs(node) {

    if (path.has(node))
        return true;

    if (visited.has(node))
        return false;

    visited.add(node);
    path.add(node);

    for (const nei of graph.get(node)) {

        if (dfs(nei))
            return true;

    }

    path.delete(node);

    return false;

}
```

# 17. Topological Sort (DFS)

```js
const order = [];

function dfs(node) {

    visited.add(node);

    for (const nei of graph.get(node)) {

        if (!visited.has(nei))

            dfs(nei);

    }

    order.push(node);

}
```

# 18. Topological Sort (Kahn's Algorithm)

```js
const indegree =
Array(n).fill(0);
```

Count indegrees:

```js
for (const [u, v] of edges)

    indegree[v]++;
```

Queue:

```js
for (let i = 0; i < n; i++) {

    if (indegree[i] === 0)

        queue.push(i);

}
```

Process:

```js
while (queue.length) {

    const node =
        queue.shift();

    for (const nei of graph.get(node)) {

        indegree[nei]--;

        if (indegree[nei] === 0)

            queue.push(nei);

    }

}
```

# 23. Find Parent

```js
function find(x) {

    if (parent[x] !== x)

        parent[x] =
            find(parent[x]);

    return parent[x];

}
```

---

# 24. Union

```js
function union(a, b) {

    const pa = find(a);
    const pb = find(b);

    if (pa !== pb)

        parent[pa] = pb;

}
```
